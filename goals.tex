\section{Directed Service}
\SectionGuidance{Describe future goals for your assigned responsibilities, with intended outcomes and indicators of progress.}

\LibraryEntry
  {}
  {Advance CSP Strategic Plan Implementation}
  {}
  {}
  {}
  {
  \begin{itemize}
    \item Lead the execution of CSP's 2025--2030 strategic plan.
    \item Expand OATI infrastructure to support increased monograph and alternative textbook publishing resulting in 50\% more year on year New Prairie Press monographs
  \end{itemize}
  }

\LibraryEntry
  {}
  {Support Library Research Compliance \& Infrastructure}
  {}
  {}
  {}
  {
  \begin{itemize}
    \item Implement an institutional data repository aligned with FAIR and Nelson Memo mandates such as Figshare or Dataverse
    \item Serve as library liaison to the Office of Research, contributing to CRIS/RIMS planning and deployment, resulting in at least two Library–OVPR initiatives
  \end{itemize}
  }

\LibraryEntry
  {}
  {Coordinate Scholarly Publishing Partnerships}
  {}
  {}
  {}
  {
  \begin{itemize}
    \item Build statewide and national partnerships around shared publishing infrastructure in the form of a Kansas Library Publishing Initiative including at least 5 colleges and universities, meeting once a month, and reduce costs by 20\% across members
    \item Engage with the Janeway/Fulcrum Coalition and support agronomy, extension, and open science publication initiatives with two additional Science oriented New Prairie Press Journals
  \end{itemize}
  }
