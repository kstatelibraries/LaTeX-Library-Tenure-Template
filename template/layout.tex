% Libraries layout, implemented with standard LaTeX packages.
% Entries remain ordinary paragraphs so long content can span pages.
\newcommand{\PortfolioSection}[1]{%
  \clearpage\phantomsection\addcontentsline{toc}{section}{#1}%
  {\raggedright\sffamily\color{RoyalPurple}\fontsize{20}{24}\selectfont #1\par}%
  {\color{RoyalPurple}\rule{0.12\textwidth}{2pt}\hspace{0.015\textwidth}%
   \color{RoyalPurple!20}\rule{0.865\textwidth}{0.4pt}\par}\vspace{0.4em}}

\newcommand{\LibraryEntry}[6]{%
  \par\addvspace{0.6em}\Needspace{7\baselineskip}%
  {\sffamily\bfseries\color{RoyalPurple}#2\par}%
  \if\relax\detokenize{#1}\relax\else {\small\color{RoyalPurple}#1\par}\fi
  \if\relax\detokenize{#3}\relax\else #3\par\fi
  \if\relax\detokenize{#4}\relax\else #4\par\fi
  \if\relax\detokenize{#5}\relax\else #5\par\fi
  #6\par}
